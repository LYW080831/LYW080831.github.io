# ============================================================
# 北京二手房数据采集与可视化
# 数据源：https://bj.ke.com/ershoufang/
# 依赖：pip install requests beautifulsoup4 pandas matplotlib
# ============================================================

import requests
from bs4 import BeautifulSoup
import pandas as pd
import matplotlib.pyplot as plt
import matplotlib
import time
import random

# ---------- 中文字体配置 ----------
matplotlib.rcParams['font.sans-serif'] = ['SimHei', 'Microsoft YaHei']
matplotlib.rcParams['axes.unicode_minus'] = False

# ---------- 配置 ----------
BASE_URL = "https://bj.ke.com/ershoufang/pg{}/"
HEADERS = {
    "User-Agent": (
        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
        "AppleWebKit/537.36 (KHTML, like Gecko) "
        "Chrome/125.0.0.0 Safari/537.36"
    ),
    "Referer": "https://bj.ke.com/ershoufang/",
    "Accept-Language": "zh-CN,zh;q=0.9",
}
TARGET_COUNT = 100   # 目标采集数量


def parse_price(text):
    """提取数字：'485万' -> 485.0, '6.8万/平' -> 68000"""
    if not text:
        return None
    import re
    nums = re.findall(r"[\d.]+", text)
    return float(nums[0]) if nums else None


def crawl_page(page_num):
    """抓取单页房源列表，返回 list[dict]"""
    url = BASE_URL.format(page_num)
    try:
        resp = requests.get(url, headers=HEADERS, timeout=15)
        resp.encoding = "utf-8"
        if resp.status_code != 200:
            print(f"  页面 {page_num} 状态码 {resp.status_code}")
            return []
    except Exception as e:
        print(f"  请求异常: {e}")
        return []

    soup = BeautifulSoup(resp.text, "lxml")
    items = soup.select("ul.sellListContent > li")
    houses = []

    for li in items:
        try:
            title_el = li.select_one("div.title a")
            title = title_el.text.strip() if title_el else ""

            # 总价
            total_el = li.select_one("div.totalPrice span")
            total = parse_price(total_el.text) if total_el else None

            # 单价
            unit_el = li.select_one("div.unitPrice span")
            unit = parse_price(unit_el.text) if unit_el else None

            # 户型 / 面积 / 朝向 / 楼层 / 装修
            info_el = li.select_one("div.houseInfo")
            info = info_el.text.strip() if info_el else ""
            parts = [p.strip() for p in info.split("|")]
            layout = parts[0] if len(parts) > 0 else ""
            area = parse_price(parts[1]) if len(parts) > 1 else None
            orient = parts[2] if len(parts) > 2 else ""
            floor = parts[3] if len(parts) > 3 else ""
            deco = parts[4] if len(parts) > 4 else ""

            # 区域 / 商圈
            pos_el = li.select_one("div.positionInfo")
            pos_text = pos_el.text.strip() if pos_el else ""
            pos_parts = pos_text.split("-")
            district = pos_parts[0].strip() if pos_parts else ""
            area_name = pos_parts[1].strip() if len(pos_parts) > 1 else ""

            houses.append({
                "标题": title, "总价": total, "单价": unit,
                "户型": layout, "面积": area, "朝向": orient,
                "楼层": floor, "装修": deco,
                "行政区": district, "商圈": area_name,
            })
        except Exception:
            continue

    return houses


def main():
    all_data = []
    page = 1

    while len(all_data) < TARGET_COUNT and page <= 10:
        print(f"正在采集第 {page} 页...")
        page_data = crawl_page(page)
        all_data.extend(page_data)
        print(f"  本页 {len(page_data)} 条，累计 {len(all_data)} 条")
        page += 1
        time.sleep(random.uniform(2, 4))  # 随机延迟防反爬

    df = pd.DataFrame(all_data[:TARGET_COUNT])
    df.to_csv("beijing_ershoufang.csv", index=False, encoding="utf-8-sig")
    print(f"\n采集完成，共 {len(df)} 条，已保存至 beijing_ershoufang.csv")

    # ---------- 可视化 ----------
    visualize(df)


def visualize(df):
    """生成扇形图、折线图、柱状图"""
    fig, axes = plt.subplots(1, 3, figsize=(18, 6))

    # ----- 1. 扇形图：各行政区房源数量占比 -----
    district_counts = df["行政区"].value_counts()
    axes[0].pie(
        district_counts.values,
        labels=district_counts.index,
        autopct="%1.1f%%",
        startangle=90,
        colors=plt.cm.Set3.colors,
    )
    axes[0].set_title("各行政区房源数量占比", fontsize=14, fontweight="bold")

    # ----- 2. 折线图：各区域平均单价走势 -----
    avg_price = df.groupby("行政区")["单价"].mean().sort_values(ascending=False)
    axes[1].plot(
        avg_price.index, avg_price.values,
        marker="o", linewidth=2, color="#e74c3c",
        markersize=8, markerfacecolor="white", markeredgewidth=2,
    )
    axes[1].set_title("各区域平均单价走势", fontsize=14, fontweight="bold")
    axes[1].set_xlabel("行政区")
    axes[1].set_ylabel("单价（元/㎡）")
    axes[1].tick_params(axis="x", rotation=45)
    axes[1].grid(True, linestyle="--", alpha=0.4)

    # ----- 3. 柱状图：各户型平均总价对比 -----
    layout_avg = df.groupby("户型")["总价"].mean().sort_values(ascending=False).head(8)
    axes[2].bar(
        layout_avg.index, layout_avg.values,
        color=plt.cm.viridis(range(len(layout_avg))),
        edgecolor="white", linewidth=0.5,
    )
    axes[2].set_title("各户型平均总价对比", fontsize=14, fontweight="bold")
    axes[2].set_xlabel("户型")
    axes[2].set_ylabel("总价（万元）")
    axes[2].tick_params(axis="x", rotation=30)
    axes[2].grid(True, axis="y", linestyle="--", alpha=0.4)

    plt.tight_layout()
    plt.savefig("beijing_ershoufang_charts.png", dpi=150, bbox_inches="tight")
    plt.show()


if __name__ == "__main__":
    main()
